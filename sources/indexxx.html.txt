<div class="blog-wrapper" style="text-align: center; background: #0b0a0c; padding: 20px;">
  <canvas id="darkFantasyBlog" width="800" height="1400" style="border: 2px solid #3a3335; box-shadow: 0 0 20px #000;"></canvas>
</div>

<script>
const canvas = document.getElementById('darkFantasyBlog');
const ctx = canvas.getContext('2d');

// --- 1. ОТРИСОВКА ШАПКИ (Header) ---
function drawHeader() {
  // Фон шапки (темное дерево / замок)
  let headerGrad = ctx.createLinearGradient(0, 0, 0, 250);
  headerGrad.addColorStop(0, '#0a080c');
  headerGrad.addColorStop(1, '#1c1618');
  ctx.fillStyle = headerGrad;
  ctx.fillRect(0, 0, 800, 250);

  // Ниша/Рамка под портрет кота
  ctx.strokeStyle = '#c5a059';
  ctx.lineWidth = 3;
  ctx.strokeRect(30, 25, 180, 200);

  // Заглушка под портрет Белого Антро-Кота в широкополой шляпе
  ctx.fillStyle = '#151316';
  ctx.fillRect(32, 27, 176, 196);
  
  /* ВСТАВКА РЕАЛЬНОГО ИЗОБРАЖЕНИЯ КОТА:
     const catImg = new Image();
     catImg.src = 'ССЫЛКА_НА_АВАТАР';
     catImg.onload = () => ctx.drawImage(catImg, 32, 27, 176, 196);
  */
  
  // Временная визуализация кота (заглушка)
  ctx.fillStyle = '#e6e6e6';
  ctx.beginPath();
  ctx.arc(120, 140, 45, 0, Math.PI * 2); // Голова кота
  ctx.fill();
  // Тень шляпы
  ctx.fillStyle = '#1a191d';
  ctx.beginPath();
  ctx.ellipse(120, 105, 65, 15, 0, 0, Math.PI * 2); // Поля шляпы
  ctx.fill();

  // Заголовок блога
  ctx.font = 'bold 28px Georgia, serif';
  ctx.fillStyle = '#c5a059';
  ctx.shadowColor = '#000';
  ctx.shadowBlur = 4;
  ctx.fillText("ХРОНИКИ ТЕМНОЙ ЛОЩИНЫ", 240, 70);

  // Подзаголовок / Краткое описание
  ctx.font = 'italic 15px Georgia, serif';
  ctx.fillStyle = '#a39274';
  ctx.fillText("Записки странника в черной шляпе • Пристанище ремесленников", 240, 100);

  // Текстурный разделитель под шапкой
  drawDivider(240, 120, 520);
}

// Декоративный разделитель
function drawDivider(x, y, width) {
  ctx.strokeStyle = '#c5a059';
  ctx.lineWidth = 1;
  ctx.beginPath();
  ctx.moveTo(x, y);
  ctx.lineTo(x + width, y);
  ctx.stroke();
  
  // Ромб по центру
  ctx.fillStyle = '#8b2626';
  ctx.beginPath();
  ctx.arc(x + width / 2, y, 4, 0, Math.PI * 2);
  ctx.fill();
}

// --- 2. БЛОК: ТЕКСТ-ИСТОРИЯ (Scroll / Lore) ---
function drawStoryBlock(y) {
  // Подложка в виде старинного свитка
  ctx.fillStyle = '#181413';
  ctx.fillRect(40, y, 720, 220);
  ctx.strokeStyle = '#4a3f35';
  ctx.strokeRect(40, y, 720, 220);

  // Заголовок секции
  ctx.font = 'bold 20px Georgia, serif';
  ctx.fillStyle = '#d4c29a';
  ctx.fillText("📜 Легенда о страннике", 60, y + 35);

  // Текст истории
  ctx.font = '14px Georgia, serif';
  ctx.fillStyle = '#b8a890';
  const lines = [
    "В ночной тишине, где лишь шорох старых фолиантов нарушает покой, шагает тень.",
    "Белый шелк шерсти скрыт под полами темного плаща, а широкие поля шляпы прячут",
    "взгляд, полный древней магии и авантюризма. Каждая запись в этом журнале —",
    "след пройденного тропа, бросок судьбоносных костей и память о верных соратниках."
  ];

  lines.forEach((line, index) => {
    ctx.fillText(line, 60, y + 75 + (index * 24));
  });
}

// --- 3. БЛОК: ПОДАРКИ-АРТЫ (Gallery Grid) ---
function drawArtsBlock(y) {
  ctx.font = 'bold 20px Georgia, serif';
  ctx.fillStyle = '#c5a059';
  ctx.fillText("🎨 Сокровищница (Арт-Подарки)", 40, y);

  drawDivider(40, y + 10, 720);

  // Сетка под галерею (3 карточки)
  const cardWidth = 220;
  const cardHeight = 150;
  const gap = 30;

  for (let i = 0; i < 3; i++) {
    let x = 40 + i * (cardWidth + gap);
    
    // Рамка карточки
    ctx.fillStyle = '#141215';
    ctx.fillRect(x, y + 30, cardWidth, cardHeight);
    ctx.strokeStyle = '#3a3335';
    ctx.strokeRect(x, y + 30, cardWidth, cardHeight);

    // Место под арт
    ctx.font = '12px Georgia, serif';
    ctx.fillStyle = '#665a4e';
    ctx.fillText(`[ Артефакт #${i + 1} ]`, x + 65, y + 100);
  }
}

// --- 4. БЛОК: ПОДАРКИ-СТИХИ (Scroll Text Blocks) ---
function drawPoemsBlock(y) {
  ctx.font = 'bold 20px Georgia, serif';
  ctx.fillStyle = '#c5a059';
  ctx.fillText("🗡️ Баллады и Сказания", 40, y);

  drawDivider(40, y + 10, 720);

  // 2 столбца под стихи
  const blockWidth = 345;
  for (let i = 0; i < 2; i++) {
    let x = 40 + i * (blockWidth + 30);

    ctx.fillStyle = '#121014';
    ctx.fillRect(x, y + 30, blockWidth, 180);
    ctx.strokeStyle = '#2d2729';
    ctx.strokeRect(x, y + 30, blockWidth, 180);

    // Текст стиха
    ctx.font = 'italic 13px Georgia, serif';
    ctx.fillStyle = '#a39274';
    ctx.fillText("«Клинок чеканит верный шаг,", x + 20, y + 60);
    ctx.fillText("В тени шляпы скрыт наш маг.", x + 20, y + 85);
    ctx.fillText("Пусть выпадет на кубе 'двадцать',", x + 20, y + 110);
    ctx.fillText("Чтоб с тьмою весело сражаться...»", x + 20, y + 135);

    ctx.font = '11px Georgia, serif';
    ctx.fillStyle = '#c5a059';
    ctx.fillText("— Песнь барда", x + 220, y + 165);
  }
}

// --- 5. БЛОК: ПОЗДРАВЛЕНИЯ И ШАБЛОН (Greetings & Guestbook) ---
function drawGreetingsBlock(y) {
  ctx.font = 'bold 20px Georgia, serif';
  ctx.fillStyle = '#c5a059';
  ctx.fillText("📜 Свитки Пожеланий", 40, y);

  drawDivider(40, y + 10, 720);

  // Шаблон для отправки/оставления поздравления
  ctx.fillStyle = '#1c1817';
  ctx.fillRect(40, y + 30, 720, 140);
  ctx.strokeStyle = '#5e4e3c';
  ctx.strokeRect(40, y + 30, 720, 140);

  ctx.font = 'bold 14px Georgia, serif';
  ctx.fillStyle = '#d4c29a';
  ctx.fillText("Оставить запись в фолианте (Шаблон поздравления):", 60, y + 55);

  ctx.font = '13px Monospace, monospace';
  ctx.fillStyle = '#8c7d6b';
  ctx.fillText("Имя странника: [Твой Ник]", 60, y + 85);
  ctx.fillText("Послание: [Да сопутствует тебе удача в бросках и верные соратники!]", 60, y + 110);
  ctx.fillText("Подарок: [Арт / Стих / Золото]", 60, y + 135);
}

// --- ИНИЦИАЛИЗАЦИЯ ВСЕЙ СЦЕНЫ ---
function renderBlog() {
  // Главный фон холста
  ctx.fillStyle = '#0f0d11';
  ctx.fillRect(0, 0, canvas.width, canvas.height);

  drawHeader();
  drawStoryBlock(270);
  drawArtsBlock(530);
  drawPoemsBlock(760);
  drawGreetingsBlock(1010);
}

renderBlog();
</script>
